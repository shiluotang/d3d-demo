cmake_minimum_required(VERSION 3.10)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 98)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(WIN32)
    set(MINGW64_HOME "C:/Dev-Cpp/MinGW64")
    set(CMAKE_C_COMPILER "${MINGW64_HOME}/bin/gcc.exe")
    set(CMAKE_CXX_COMPILER "${MINGW64_HOME}/bin/g++.exe")
else()
    set(CMAKE_C_COMPILER "/usr/bin/x86_64-w64-mingw32-gcc")
    set(CMAKE_CXX_COMPILER "/usr/bin/x86_64-w64-mingw32-g++")
endif()

add_library(d3d9 IMPORTED STATIC)
add_library(d3dx9_43 IMPORTED STATIC)
add_library(glu32 IMPORTED STATIC)
if(WIN32)
    set_target_properties(d3d9 PROPERTIES IMPORTED_LOCATION "${MINGW64_HOME}/x86_64-w64-mingw32/lib/libd3d9.a")
    set_target_properties(d3dx9_43 PROPERTIES IMPORTED_LOCATION "${MINGW64_HOME}/x86_64-w64-mingw32/lib/libd3dx9_43.a")
    set_target_properties(glu32 PROPERTIES IMPORTED_LOCATION "${MINGW64_HOME}/x86_64-w64-mingw32/lib/libglu32.a")
else()
    set_target_properties(d3d9 PROPERTIES IMPORTED_LOCATION "/usr/x86_64-w64-mingw32/lib/libd3d9.a")
    set_target_properties(d3dx9_43 PROPERTIES IMPORTED_LOCATION "/usr/x86_64-w64-mingw32/lib/libd3dx9_43.a")
    set_target_properties(glu32 PROPERTIES IMPORTED_LOCATION "/usr/x86_64-w64-mingw32/lib/libglu32.a")
endif()

project(d3d-demo VERSION 1.0)
add_subdirectory(src)
